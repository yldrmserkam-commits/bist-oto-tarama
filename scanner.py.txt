tickers = list(set([t.replace('.IS', '') for t in ham_tickers]))

print("Tarama başlatılıyor...")
for periyot_adi, aktif_mi in TARAMA_YAPILACAK_PERIYOTLAR.items():
    if not aktif_mi:
        continue

    ayar = PERIYOT_AYARLARI[periyot_adi]

    for ticker in tickers:
        try:
            ticker_symbol = f"{ticker}.IS"
            df = yf.download(ticker_symbol, period=ayar["period"], interval=ayar["interval"], progress=False)

            if df.empty or len(df) < max(CCI_PERIYOT + 5, 10):
                continue

            if isinstance(df.columns, pd.MultiIndex):
                df.columns = df.columns.droplevel(1)

            if ayar["resample_rule"]:
                df = df.resample(ayar["resample_rule"]).agg({
                    'Open': 'first', 'High': 'max', 'Low': 'min', 'Close': 'last', 'Volume': 'sum'
                }).dropna()

                if len(df) < max(CCI_PERIYOT + 5, 10):
                    continue

            if float(df['Volume'].iloc[-1]) == 0:
                continue

            ema6 = df['Close'].ewm(span=6, adjust=False).mean()
            close_curr = float(df['Close'].iloc[-1])
            close_prev = float(df['Close'].iloc[-2])
            ema6_curr = float(ema6.iloc[-1])
            ema6_prev = float(ema6.iloc[-2])

            ema6_cross_above = (close_curr > ema6_curr) and (close_prev <= ema6_prev)
            if not ema6_cross_above:
                continue

            tp = (df['High'] + df['Low'] + df['Close']) / 3
            sma_tp = tp.rolling(window=CCI_PERIYOT).mean()
            mad = tp.rolling(window=CCI_PERIYOT).apply(lambda x: np.abs(x - x.mean()).mean(), raw=True)
            cci = (tp - sma_tp) / (0.015 * mad)

            prev_cci = float(cci.iloc[-2])
            curr_cci = float(cci.iloc[-1])
            cci_cross_above_minus100 = (prev_cci < -100) and (curr_cci >= -100)

            if cci_cross_above_minus100:
                # Mum zamanını kontrol et (Aynı mum için tekrar bildirim gitmesin)
                last_candle = str(df.index[-1])
                signal_key = f"{ticker}_{periyot_adi}"

                if sent_signals.get(signal_key) == last_candle:
                    continue  # Bu mum için zaten bildirim gönderilmiş

                tv_link = f"https://www.tradingview.com/chart/?symbol=BIST:{ticker}"
                msg = (
                    f"🚨 *SİNYAL YAKALANDI!*\n"
                    f"*Hisse:* `{ticker}`\n"
                    f"*Periyot:* {periyot_adi}\n"
                    f"*Fiyat:* {close_curr}\n"
                    f"*CCI:* {curr_cci:.2f}\n\n"
                    f"📈 [TradingView Grafiği Aç]({tv_link})"
                )
                telegram_mesaj_gonder(msg)

                # Hafızaya kaydet
                sent_signals[signal_key] = last_candle

            time.sleep(0.01)
        except Exception:
            pass

# Hafıza dosyasını güncelle
