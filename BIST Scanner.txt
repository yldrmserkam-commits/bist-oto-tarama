name: BIST Scanner

on:
  schedule:
    - cron: '0 6-15 * * 1-5'
  workflow_dispatch:

jobs:
  run-scanner:
     runs-on: ubuntu-latest
     steps:
       - name: Checkout repository
         uses: actions/checkout@v4

       - name: Set up Python
         uses: actions/setup-python@v5
         with:
           python-version: '3.10'

       - name: Install dependencies
         run: |
           pip install yfinance pandas numpy requests openpyxl

       - name: Run scanner script
         run: |
           python scanner.py
